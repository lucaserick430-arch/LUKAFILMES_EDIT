app.get("/api/acesso-assistir", async (req, res) => {

    try {

        if (!req.session || !req.session.usuario) {
            return res.status(401).json({
                permitido: false,
                logado: false,
                motivo: "nao_logado"
            });
        }

        const sessao = req.session.usuario;
        const usuarios = await carregarUsuarios();

        const pessoa = usuarios.find(
            u => Number(u.id) === Number(sessao.id)
        );

        if (!pessoa) {
            return res.status(404).json({
                permitido: false,
                logado: true,
                motivo: "usuario_nao_encontrado"
            });
        }

        const tipo = String(pessoa.tipo || "usuario").toLowerCase();
        const status = String(pessoa.status || "ativo").toLowerCase();

        if (status !== "ativo") {
            return res.json({
                permitido: false,
                logado: true,
                motivo: "inativo",
                mensagem: "Seu acesso está desativado."
            });
        }

        if (
            tipo === CONFIGURACAO_ACESSO_LUKAFILMES.tipos.ADMIN ||
            tipo === CONFIGURACAO_ACESSO_LUKAFILMES.tipos.ISENTO
        ) {
            return res.json({
                permitido: true,
                logado: true,
                tipo,
                validade: pessoa.validade || null
            });
        }

        if (
            pessoa.validade &&
            new Date(pessoa.validade).getTime() > Date.now()
        ) {
            return res.json({
                permitido: true,
                logado: true,
                tipo,
                validade: pessoa.validade
            });
        }

        return res.json({
            permitido: false,
            logado: true,
            tipo,
            motivo: "expirado",
            valor: CONFIGURACAO_ACESSO_LUKAFILMES.valorMensal,
            dias: CONFIGURACAO_ACESSO_LUKAFILMES.diasAcesso,
            validade: pessoa.validade || null
        });

    } catch (erro) {

        console.error(
            "[ACESSO ASSISTIR] Erro:",
            erro
        );

        return res.status(500).json({
            permitido: false,
            logado: true,
            motivo: "erro_servidor",
            mensagem: "Erro interno ao verificar o acesso."
        });

    }

});

app.get("/api/admin/usuarios", async (req, res) => {
    try {
        if (
            !req.session.usuario ||
            req.session.usuario.tipo !== "admin"
        ) {
            return res.status(403).json({
                sucesso: false,
                mensagem: "Acesso negado."
            });
        }

        const usuarios = await carregarUsuarios();

        const lista = usuarios.map(u => ({
            id: u.id,
            usuario: u.usuario,
            tipo: u.tipo || "usuario",
            status: u.status || "ativo",
            validade: u.validade || null,
            criado_em: u.criado_em || null,
            online: Boolean(u.online)
        }));

        return res.json({
            sucesso: true,
            usuarios: lista,
            total: lista.length
        });

    } catch (erro) {
        console.error("[ERRO LISTAR USUARIOS]", erro);

        return res.status(500).json({
            sucesso: false,
            mensagem: "Erro interno ao carregar usuários.",
            usuarios: []
        });
    }
});

app.post(
    "/api/admin/usuarios",
    async (req, res) => {

        try {

            if (
                !req.session.usuario ||
                req.session.usuario.tipo !== "admin"
            ) {

                return res.status(403).json({
                    sucesso: false,
                    mensagem: "Acesso negado."
                });

            }

            const usuario =
                String(req.body.usuario || "").trim();

            const senha =
                String(req.body.senha || "");

            const dias =
                Number(req.body.dias);

            if (
                !usuario ||
                !senha ||
                !Number.isInteger(dias) ||
                dias < 1
            ) {

                return res.json({
                    sucesso: false,
                    mensagem:
                        "Preencha usuário, senha e dias corretamente."
                });

            }

            const usuarios =
                await carregarUsuarios();

            const existente =
                usuarios.find(
                    u =>
                        String(u.usuario || "").toLowerCase() ===
                        usuario.toLowerCase()
                );

            if (existente) {

                return res.json({
                    sucesso: false,
                    mensagem:
                        "Esse usuário já existe."
                });

            }

            const senhaHash =
                await bcrypt.hash(
                    senha,
                    12
                );

            const validade =
                new Date(
                    Date.now() +
                    dias *
                    24 *
                    60 *
                    60 *
                    1000
                );

            const novoUsuario = {

                id: proximoId(usuarios),

                usuario,

                senha: senhaHash,

                status: "ativo",

                tipo: "usuario",

                validade:
                    validade.toISOString(),

                criado_em:
                    new Date().toISOString()

            };

            usuarios.push(novoUsuario);

            await salvarUsuarios(usuarios);

            console.log(
                "[ADMIN] Usuário criado:",
                usuario
            );

            return res.json({

                sucesso: true,

                mensagem:
                    "Usuário criado com sucesso."

            });

        } catch (erro) {

            console.error(
                "[ERRO CRIAR USUARIO]",
                erro
            );

            return res.status(500).json({

                sucesso: false,

                mensagem:
                    "Erro interno ao criar usuário."

            });

        }

    }
);

async function salvar(e){

 e.preventDefault();

 const tipo=document.getElementById("tipoCriacao").value;

 const autoBox=document.getElementById("lukaAutomaticoBox");

 if(tipo==="teste" && autoBox && autoBox.style.display==="block"){
  await LUKA_CRIAR_TESTES_AUTOMATICOS();
  return;
 }

 const usuario=document.getElementById("fUsuario").value.trim();
 const senha=document.getElementById("fSenha").value;
 const dias=Number(document.getElementById("fDias").value);

 const valorInput=document.getElementById("fValor");
 const telefoneInput=document.getElementById("fTelefone");

 const valor=valorInput
  ? Number(valorInput.value || 0)
  : 0;

 const telefone=telefoneInput
  ? telefoneInput.value.trim()
  : "";

 if(
  !usuario ||
  !senha ||
  !Number.isInteger(dias) ||
  dias<1 ||
  !Number.isFinite(valor) ||
  valor<0
 ){
  toast("Preencha os dados corretamente.");
  return;
 }

 const btn=document.getElementById("btnSalvar");
 btn.disabled=true;
 btn.textContent="Criando...";

 try{

  let url="/api/admin/usuarios";
  let body={usuario,senha,dias,valor,telefone};

  if(tipo==="revendedor"){
   url="/api/admin/revendedores";
   body.limite_conexoes=Number(document.getElementById("fLimite").value);
  }

  /*
   O endpoint de usuário existente cria uma conta normal.
   O teste será tratado pelo servidor quando a rota específica
   estiver disponível.
  */
  if(tipo==="teste"){
   url="/api/admin/usuarios/teste";
  }

  const r=await fetch(url,{
   method:"POST",
   credentials:"include",
   headers:{"Content-Type":"application/json"},
   body:JSON.stringify(body)
  });

  const d=await r.json().catch(()=>({}));

  if(!r.ok||d.sucesso===false){
   throw new Error(d.mensagem||"Não foi possível criar.");
  }

  fecharModal();

  mostrarMensagemCriacao(
   d,
   {
    usuario,
    senha,
    valor,
    telefone
   }
  );

function atualizarGraficoTempoReal(){

    const lista =
        Array.isArray(usuarios)
            ? usuarios
            : [];

    const totalUsuarios =
        lista.filter(u => u.tipo === "usuario").length;

    const totalTestes =
        lista.filter(u => u.tipo === "teste").length;

    const totalOnline =
        lista.filter(u => Boolean(u.online)).length;

    const totalAtivos =
        lista.filter(u =>
            String(u.status || "ativo").toLowerCase() === "ativo"
        ).length;

    const totalExpirados =
        lista.filter(u => {
            if(!u.validade) return false;
            const d = new Date(u.validade);
            return !Number.isNaN(d.getTime()) &&
                   d.getTime() <= Date.now();
        }).length;

    const valores = [
        totalUsuarios,
        totalTestes,
        totalOnline,
        totalAtivos,
        totalExpirados,
        totalOnline + totalTestes,
        totalUsuarios + totalOnline,
        lista.length
    ];

    const maior =
        Math.max(...valores, 1);

    const barras =
        document.querySelectorAll("#lukaLiveBars div");

    barras.forEach((barra, i) => {

        const valor =
            valores[i] ?? 0;

        const altura =
            Math.max(
                12,
                Math.min(
                    100,
                    (valor / maior) * 100
                )
            );

        barra.style.height =
            altura + "%";
    });
}

function iniciarGraficoTempoReal(){

    atualizarGraficoTempoReal();

    if(window.__lukaGraficoTimer){
        clearInterval(window.__lukaGraficoTimer);
    }

    window.__lukaGraficoTimer =
        setInterval(() => {

            atualizarGraficoTempoReal();

        }, 3000);
}

function formatarDataHoraConta(v){

    if(!v){
        return "Não informado";
    }

    const d = new Date(v);

    if(Number.isNaN(d.getTime())){
        return "Não informado";
    }

    return d.toLocaleString("pt-BR",{
        dateStyle:"short",
        timeStyle:"short"
    });
}

function calcularTempoRestante(v){

    if(!v){
        return "Sem validade";
    }

    const fim = new Date(v).getTime();

    if(!Number.isFinite(fim)){
        return "Data inválida";
    }

    let restante = fim - Date.now();

    if(restante <= 0){
        return "Expirado";
    }

    const totalSegundos = Math.floor(restante / 1000);

    const dias = Math.floor(totalSegundos / 86400);
    const horas = Math.floor((totalSegundos % 86400) / 3600);
    const minutos = Math.floor((totalSegundos % 3600) / 60);

    if(dias > 0){
        return `${dias} dia${dias===1?"":"s"}, ${horas}h`;
    }

    if(horas > 0){
        return `${horas}h ${minutos}min`;
    }

    return `${Math.max(1,minutos)} min`;
}

function copiarTelefoneConta(valor){

    if(!valor){
        toast("Telefone não informado.");
        return;
    }

    if(navigator.clipboard){
        navigator.clipboard.writeText(String(valor))
            .then(() => toast("📋 Telefone copiado!"))
            .catch(() => copiarTelefoneFallback(valor));
    }else{
        copiarTelefoneFallback(valor);
    }
}

function copiarTelefoneFallback(valor){

    const area = document.createElement("textarea");
    area.value = String(valor);
    area.style.position = "fixed";
    area.style.opacity = "0";

    document.body.appendChild(area);
    area.focus();
    area.select();

    try{
        document.execCommand("copy");
        toast("📋 Telefone copiado!");
    }catch(e){
        toast("Não foi possível copiar.");
    }

    area.remove();
}
