async function LUKA_TESTAR_COMO_USUARIO() {

    const botao = document.querySelector(
        '#lukaTesteUsuarioAdmin button'
    );

    if (botao) {
        botao.disabled = true;
        botao.textContent = "🧪 INICIANDO TESTE...";
    }

    try {

        const resposta = await fetch(
            "/api/admin/teste/iniciar",
            {
                method: "POST",
                headers: {
                    "Content-Type": "application/json"
                }
            }
        );

        const dados = await resposta.json();

        if (!dados.sucesso) {
            throw new Error(
                dados.mensagem || "Não foi possível iniciar o teste."
            );
        }

        /*
         * Nova aba:
         * a sessão ADMIN continua existindo no servidor.
         * O cookie luka_modo_teste seleciona o usuário de teste
         * somente nas rotas que reconhecem esse modo.
         */
        window.open(
            dados.destino || "/?luka_teste=1",
            "_blank"
        );

    } catch (erro) {

        console.error("[LUKA TESTE]", erro);

        if (typeof toast === "function") {
            toast(erro.message);
        } else {
            alert(erro.message);
        }

    } finally {

        if (botao) {
            botao.disabled = false;
            botao.textContent = "🧪 TESTAR COMO USUÁRIO";
        }
    }
}
</script>
