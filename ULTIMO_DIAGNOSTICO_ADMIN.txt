function toggleSearch(){
    const box=document.getElementById("searchBox");
    box.classList.toggle("show");

    if(box.classList.contains("show")){
        setTimeout(()=>{
            document.getElementById("searchInput").focus();
        },100);
    }
}

function toggleMenu(){

    /*
     * CELULAR:
     * mantém exatamente o menu mobile atual.
     */
    if(window.innerWidth <= 768){

        document
            .getElementById("mobileMenu")
            .classList
            .toggle("show");

        return;
    }

    /*
     * COMPUTADOR:
     * abre somente o menu de usuário.
     */
    const menu =
        document.getElementById("desktopUserMenu");

    if(!menu){
        return;
    }

    menu.classList.toggle("show");

    if(
        menu.classList.contains("show") &&
        !menu.dataset.carregado
    ){
        carregarMenuUsuarioDesktop();
    }
}

async function LUKA_CRIAR_TESTES_AUTOMATICOS(){

 const horas=Number(document.getElementById("lukaAutoHoras").value);
 const quantidade=Number(document.getElementById("lukaAutoQuantidade").value);
 const limite_conexoes=Number(document.getElementById("lukaAutoConexoes").value);

  toast("Escolha uma duração entre 1 e 72 horas.");
  return;

  toast("A quantidade deve estar entre 1 e 100 testes.");

   method:"POST",
   credentials:"include",
   headers:{"Content-Type":"application/json"},
   body:JSON.stringify({
    horas,
    quantidade,
    limite_conexoes
   })
  });

  const d=await r.json().catch(()=>({}));

  if(!r.ok || d.sucesso===false){

    senha:"",
    valor:0,
    telefone:""
   });
  }

  if(typeof carregar==="function"){
   await carregar();
  }

 }catch(err){

  toast("❌ "+err.message);

 }finally{

  if(btn){
   btn.disabled=false;

  }
 }
