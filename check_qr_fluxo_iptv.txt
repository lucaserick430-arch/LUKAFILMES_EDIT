let lukaQRCodePromise = null;

function carregarQRCodeLuka() {

    if (typeof QRCode !== "undefined") {
        return Promise.resolve();
    }

    if (lukaQRCodePromise) {
        return lukaQRCodePromise;
    }

    lukaQRCodePromise = new Promise(function(resolve, reject) {

        const script = document.createElement("script");

        script.src =
            "https://cdn.jsdelivr.net/npm/qrcodejs@1.0.0/qrcode.min.js";

        script.onload = function() {
            resolve();
        };

        script.onerror = function() {
            lukaQRCodePromise = null;
            reject(new Error("Não foi possível carregar o QR Code."));
        };

        document.head.appendChild(script);
    });

    return lukaQRCodePromise;
}
