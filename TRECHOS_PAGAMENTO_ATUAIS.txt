        function atualizarLista(){

            const q=(busca.value||"").toLowerCase();

            const filtrada=lista.filter(u=>
                String(u.usuario||"")
                    .toLowerCase()
                    .includes(q)

                    </div>
                `;

                        detalhe+=" • expira "+

                        class="luka-central-user"
                        onclick="lukaAbrirContaCentral(${Number(u.id)})"
                        style="cursor:pointer"
                    >

                        <div class="luka-central-avatar">
