function atualizarGraficoTempoReal(){

    const lista =
        Array.isArray(usuarios)
            ? usuarios
            : [];

    const totalUsuarios =
        lista.filter(u => u.tipo === "usuario").length;

    const totalTestes =
        lista.filter(u => u.tipo === "teste").length;

    const totalOnline =
        lista.filter(u => Boolean(u.online)).length;

    const totalAtivos =
        lista.filter(u =>
            String(u.status || "ativo").toLowerCase() === "ativo"
        ).length;

    const totalExpirados =
        lista.filter(u => {
            if(!u.validade) return false;
            const d = new Date(u.validade);
            return !Number.isNaN(d.getTime()) &&
                   d.getTime() <= Date.now();
        }).length;

    const valores = [
        totalUsuarios,
        totalTestes,
        totalOnline,
        totalAtivos,
        totalExpirados,
        totalOnline + totalTestes,
        totalUsuarios + totalOnline,
        lista.length
    ];

    const maior =
        Math.max(...valores, 1);

    const barras =
        document.querySelectorAll("#lukaLiveBars div");

    barras.forEach((barra, i) => {

        const valor =
            valores[i] ?? 0;

        const altura =
            Math.max(
                12,
                Math.min(
                    100,
                    (valor / maior) * 100
                )
            );

        barra.style.height =
            altura + "%";
    });
}

function iniciarGraficoTempoReal(){

    atualizarGraficoTempoReal();

    if(window.__lukaGraficoTimer){
        clearInterval(window.__lukaGraficoTimer);
    }

    window.__lukaGraficoTimer =
        setInterval(() => {

            atualizarGraficoTempoReal();

        }, 3000);
}
