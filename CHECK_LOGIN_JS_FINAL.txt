<script>
(function () {

    function iniciarLukafilmes() {
        var botao = document.getElementById("botaoEntrar");

        if (!botao) return;

        botao.onclick = function (evento) {
            evento.preventDefault();

            if (botao.dataset.abrindo === "1") return;

            botao.dataset.abrindo = "1";
            botao.disabled = true;

            botao.innerHTML =
                '<span class="icone-play">▶</span>' +
                '<span>ABRINDO LUKAFILMES...</span>';

            try {
                if (typeof window.lukafilmesMostrarLoginLoader === "function") {
                    window.lukafilmesMostrarLoginLoader();
                }
            } catch (e) {}

            setTimeout(function () {
                window.location.href = "/";
            }, 350);
        };
    }

    if (document.readyState === "loading") {
        document.addEventListener("DOMContentLoaded", iniciarLukafilmes);
    } else {
        iniciarLukafilmes();
    }

})();
</script>
