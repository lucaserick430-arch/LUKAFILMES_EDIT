(function(){

    var loader = document.getElementById("lukafilmesLoginLoader");
    var form = document.getElementById("loginForm");

    function mostrarLoginLoader(){
        if(!loader)return;
        loader.classList.add("ativo");
        loader.setAttribute("aria-hidden","false");
    }

    function esconderLoginLoader(){
        if(!loader)return;
        loader.classList.remove("ativo");
        loader.setAttribute("aria-hidden","true");
    }

    window.lukafilmesMostrarLoginLoader = mostrarLoginLoader;
    window.lukafilmesEsconderLoginLoader = esconderLoginLoader;

    if(form){
        form.addEventListener("submit",function(){
            mostrarLoginLoader();
        },true);
    }

    window.addEventListener("pageshow",function(){
        esconderLoginLoader();
    });

})();
</script>
