<script id="lukafilmes-pesquisa-home-final">
(function () {

    const input = document.getElementById("searchInput");
    const box = document.getElementById("searchBox");

    if (!input || !box) {
        console.warn("[PESQUISA HOME] Campo não encontrado.");
        return;
    }

    let area = document.getElementById("resultadosBuscaHome");

    if (!area) {
        area = document.createElement("div");
        area.id = "resultadosBuscaHome";
        box.appendChild(area);
    }

    let contador = 0;

    function escapar(valor) {
        return String(valor || "")
            .replace(/&/g, "&amp;")
            .replace(/</g, "&lt;")
            .replace(/>/g, "&gt;")
            .replace(/"/g, "&quot;")
            .replace(/'/g, "&#039;");
    }

    function esconder() {
        area.innerHTML = "";
        area.style.display = "none";
    }

    async function pesquisarHome(termo) {

        const pesquisaAtual = ++contador;

        if (!termo) {
            esconder();
            return;
        }

        area.style.display = "block";
        area.innerHTML =
            '<div class="busca-home-status">🔎 Pesquisando...</div>';

        try {

            const resposta = await fetch(
                "/api/pesquisar?q=" +
                encodeURIComponent(termo)
            );

            if (!resposta.ok) {
                throw new Error("HTTP " + resposta.status);
            }

            const dados = await resposta.json();

            if (pesquisaAtual !== contador) return;

            const lista = Array.isArray(dados.resultados)
                ? dados.resultados
                : [];

            area.innerHTML = "";

            if (!lista.length) {
                area.innerHTML =
                    '<div class="busca-home-status">Nenhum resultado encontrado.</div>';
                return;
            }

            lista.slice(0, 20).forEach(function (filme) {

                const item = document.createElement("div");
                item.className = "busca-home-item";

                const capa = filme.capa || "";
                const titulo = filme.titulo || "Sem título";
                const ano = filme.ano || "";

                item.innerHTML = `
                    ${capa
                        ? `<img loading="lazy" decoding="async" src="${escapar(capa)}" alt="">`
                        : ""
                    }
                    <div class="busca-home-info">
                        <div class="busca-home-titulo">
                            ${escapar(titulo)}
                        </div>
                        <div class="busca-home-ano">
                            ${escapar(ano)}
                        </div>
                    </div>
                `;

                item.addEventListener("click", function () {

                    localStorage.setItem(
                        "filmeSelecionado",
                        JSON.stringify(filme)
                    );

                    window.location.href =
                        "/paginas/filme.html";
                });

                area.appendChild(item);
            });

        } catch (erro) {

            if (pesquisaAtual !== contador) return;

            console.error(
                "[PESQUISA HOME]",
                erro
            );

            area.innerHTML =
                '<div class="busca-home-status">Erro ao pesquisar.</div>';
        }
    }

    input.addEventListener("input", function () {
        pesquisarHome(input.value.trim());
    });

    input.addEventListener("keydown", function (event) {

        if (event.key === "Enter") {

            event.preventDefault();

            const primeiro =
                area.querySelector(".busca-home-item");

            if (primeiro) {
                primeiro.click();
            }
        }
    });

    document.addEventListener("click", function (event) {

        if (!box.contains(event.target)) {
            esconder();
        }

    });

    console.log(
        "[PESQUISA HOME] Sistema ativo."
    );

})();
</script>
