async function assistirEpisodio(
        temporada,
        episodio,
        nome
    ) {

        if (!serie || !serie.id) {

            console.error(
                "[SERIES] ID da série não encontrado."
            );

            return;
        }

        const idSerie =
            String(serie.id).trim();

        const temporadaId =
            String(temporada).trim();

        const episodioId =
            String(episodio).trim();

        if (!idSerie || !temporadaId || !episodioId) {

            console.error(
                "[SERIES] Dados do episódio incompletos.",
                {
                    idSerie,
                    temporadaId,
                    episodioId
                }
            );

            return;
        }

        /*
         * VERIFICA O ACESSO ANTES DE ABRIR O EPISÓDIO.
         * Usa a mesma rota de acesso dos filmes.
         */
        try {
            const respostaAcesso =
                await fetch(
                    "/api/acesso-assistir",
                    {
                        method: "GET",
                        credentials: "same-origin",
                        cache: "no-store"
                    }
                );

            const acesso =
                await respostaAcesso.json();

            console.log(
                "[LUKAFILMES] Acesso para assistir série:",
                acesso
            );

            if (
                acesso.motivo ===
                "nao_logado"
            ) {
                const retorno =
                    window.location.pathname +
                    window.location.search;

                window.location.href =
                    "/cadastro?retorno=" +
                    encodeURIComponent(retorno);

                return;
            }

            if (!acesso.permitido) {
                if (
                    acesso.motivo ===
                    "expirado"
                ) {
                    await mostrarPagamentoLuka(
                        acesso
                    );

                    return;
                }

                alert(
                    acesso.mensagem ||
                    "Seu acesso não está liberado para assistir."
                );

                return;
            }

        } catch (erro) {
            console.error(
                "[LUKAFILMES] Erro ao verificar acesso da série:",
                erro
            );

            alert(
                "Não foi possível verificar seu acesso. Tente novamente."
            );

            return;
        }

        console.log(
            "[SERIES PLAYER] Abrindo episódio:",
            {
                serie: idSerie,
                temporada: temporadaId,
                episodio: episodioId,
                nome: nome || ""
            }
        );

        const playerSerie =
            document.getElementById("playerSerie");

        const serieFrame =
            document.getElementById("serieFrame");

        if (!playerSerie || !serieFrame) {

            console.error(
                "[SERIES PLAYER] Área do player não encontrada."
            );

            return;
        }

        /*
         * GUARDA OS DADOS DO EPISÓDIO
         */

        playerSerie.dataset.serieId =
            idSerie;

        playerSerie.dataset.temporada =
            temporadaId;

        playerSerie.dataset.episodio =
            episodioId;

        /*
         * ABRE O PLAYER
         */

        playerSerie.style.display =
            "block";

        /*
         * PLAY 1 — MEGA
         * INICIA COMO PADRÃO
         */

        const play1Url =
            "https://mgeb.top/embed/" +
            idSerie +
            "/" +
            temporadaId +
            "/" +
            episodioId;

        serieFrame.src =
            play1Url;

        /*
         * CONTINUAR ASSISTINDO
         * Registra imediatamente o episódio clicado.
         * O episódio começa sempre no PLAY 1.
         */
        if (window.LukaRegistrarSerieContinuar) {

            window.LukaRegistrarSerieContinuar({

                id: idSerie,

                titulo:
                    serie.name ||
                    serie.titulo ||
                    nome ||
                    "Série",

                capa:
                    serie.poster_path
                        ? "https://image.tmdb.org/t/p/w500" +
                          serie.poster_path
                        : "",

                tipo: "serie",

                player: "1",

                temporada:
                    temporadaId,

                episodio:
                    episodioId

            });

            console.log(
                "[CONTINUAR SÉRIE] Episódio iniciado:",
                {
                    id: idSerie,
                    temporada: temporadaId,
                    episodio: episodioId,
                    player: "1"
                }
            );
        }

        /*
         * MARCA PLAY 1
         */

        const botoesPlayers =
            playerSerie.querySelectorAll(
                ".serie-player-btn"
            );

        botoesPlayers.forEach(function(btn) {

            btn.classList.remove(
                "ativo"
            );

            btn.classList.remove(
                "player-selecionado"
            );

        });

        const play1 =
            playerSerie.querySelector(
                '.serie-player-btn[data-player="1"]'
            );

        if (play1) {

            play1.classList.add(
                "ativo"
            );

            play1.classList.add(
                "player-selecionado"
            );

        }

        /*
         * CONFIGURA PLAY 1 E PLAY 2
         */

        botoesPlayers.forEach(function(btn) {

            if (
                btn.dataset.playerConfigurado === "1"
            ) {
                return;
            }

            btn.dataset.playerConfigurado =
                "1";

            btn.addEventListener(
                "click",
                function() {

                    const numero =
                        String(
                            this.dataset.player || ""
                        );

                    let embedUrl = "";

                    /*
                     * PLAY 1 — MEGA
                     */

                    if (numero === "1") {

                        embedUrl =
                            "https://mgeb.top/embed/" +
                            idSerie +
                            "/" +
                            temporadaId +
                            "/" +
                            episodioId;
                    }

                    /*
                     * PLAY 2 — MAYA / MYEMBED
                     */

                    else if (numero === "2") {

                        embedUrl =
                            "https://pipocacine.lat/embed/" +
                            String(idSerie).trim() +
                            "/" +
                            String(temporadaId).trim() +
                            "/" +
                            String(episodioId).trim();
                    }

                    /*
                     * PLAY 3 — APIPLAYER
                     * SÉRIE / TV
                     * IDIOMA: PORTUGUÊS
                     */

                    else if (numero === "3") {

                        embedUrl =
                            "https://myembed.biz/serie/" +
                            idSerie +
                            "/" +
                            temporadaId +
                            "/" +
                            episodioId;
                    }


                    else if (numero === "4") {

                        embedUrl =
                            "https://apiplayer.ru/embed/tv/" +
                            String(idSerie).trim() +
                            "/" +
                            String(temporadaId).trim() +
                            "/" +
                            String(episodioId).trim() +
                            "?lang=pt";
                    }

                    if (!embedUrl) {
                        return;
                    }

                    /*
                     * DESELECIONA OS OUTROS
                     */

                    botoesPlayers.forEach(
                        function(item) {

                            item.classList.remove(
                                "ativo"
                            );

                            item.classList.remove(
                                "player-selecionado"
                            );

                        }
                    );

                    /*
                     * MARCA O ESCOLHIDO
                     */

                    this.classList.add(
                        "ativo"
                    );

                    this.classList.add(
                        "player-selecionado"
                    );

                    /*
                     * TROCA O IFRAME
                     */

                    serieFrame.src =
                        embedUrl;

                    console.log(
                        "[SERIES PLAYER]",
                        "PLAY " + numero,
                        embedUrl
                    );

                    /*
                     * CONTINUAR ASSISTINDO
                     *
                     * Atualiza somente o episódio/player atual.
                     * O storage já está isolado por conta.
                     */
                    if (window.LukaRegistrarSerieContinuar) {

                        window.LukaRegistrarSerieContinuar({

                            id: idSerie,

                            titulo:
                                serie.name ||
                                serie.titulo ||
                                nome ||
                                "Série",

                            capa:
                                serie.poster_path
                                    ? "https://image.tmdb.org/t/p/w500" +
                                      serie.poster_path
                                    : "",

                            tipo: "serie",

                            player: String(numero),

                            temporada:
                                String(temporadaId),

                            episodio:
                                String(episodioId)

                        });

                        console.log(
                            "[CONTINUAR SÉRIE] PLAYER ATUALIZADO:",
                            {
                                id: idSerie,
                                temporada: String(temporadaId),
                                episodio: String(episodioId),
                                player: String(numero)
                            }
                        );
                    }

                    

                }
            );

        });

        /*
         * DESCE ATÉ O PLAYER
         */

        playerSerie.scrollIntoView({
            behavior: "smooth",
            block: "center"
        });

        console.log(
            "[SERIES PLAYER] PLAY 1 carregado:",
            play1Url
        );

    }


    /* ==========================================
       VOLTAR PARA A LISTAGEM PRESERVANDO POSIÇÃO
       ========================================== */

    const botaoVoltarSerie =
        document.getElementById("voltarSerie");

    if (botaoVoltarSerie) {

        botaoVoltarSerie.addEventListener(
            "click",
            function (e) {

                e.preventDefault();

                window.location.href = "series.html";

            }
        );

    }

    /* ==========================================
       INICIAR
       ========================================== */

    carregarDetalhes();


})();
