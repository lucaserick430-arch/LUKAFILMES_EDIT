(function () {
    "use strict";

    /*
     * LUKAFILMES — PROTEÇÃO DE NAVEGAÇÃO DO PLAYER
     *
     * Mantém a navegação iniciada pelo próprio LUKAFILMES
     * na página atual e evita que elementos externos do site
     * sejam transformados em navegação da página principal.
     */

    const frame = document.getElementById("filmeFrame");

    if (!frame) {
        console.log("[PLAYER PROTEÇÃO] filmeFrame não encontrado.");
        return;
    }

    /*
     * Garante que o iframe continue sendo o destino do player.
     */
    frame.setAttribute("allowfullscreen", "");
    frame.setAttribute(
        "allow",
        "autoplay; encrypted-media; picture-in-picture; fullscreen"
    );

    /*
     * Bloqueia links target=_blank pertencentes à página principal.
     * Não interfere nos links internos do conteúdo do iframe externo.
     */
    document.addEventListener("click", function (evento) {

        const alvo = evento.target;

        if (!alvo || !alvo.closest) return;

        const link = alvo.closest("a");

        if (!link) return;

        if (link.target === "_blank") {
            link.target = "_self";
        }
    }, true);

    /*
     * Evita window.open disparado diretamente pelo código
     * da página principal.
     */
    const abrirOriginal = window.open;

    window.open = function (url, target, features) {

        if (!url) {
            return null;
        }

        console.log(
            "[PLAYER PROTEÇÃO] tentativa de nova janela bloqueada:",
            url
        );

        return null;
    };

    console.log("[PLAYER PROTEÇÃO] ativa.");
})();
</script>


<script id="lukafilmes-player-protecao">
(function () {
    "use strict";

    /*
     * Proteção da página principal.
     * O conteúdo dos iframes externos continua isolado.
     * Não intercepta nem modifica anúncios internos dos players.
     */

    const iframe = document.getElementById("filmeFrame");

    if (!iframe) {
        console.log("[PLAYER PROTECAO] filmeFrame não encontrado.");
        return;
    }

    iframe.setAttribute(
        "referrerpolicy",
        "strict-origin-when-cross-origin"
    );

    /*
     * Mantém o iframe dentro da área do player.
     */
    iframe.addEventListener("load", function () {
        console.log(
            "[PLAYER PROTECAO] iframe carregado:",
            iframe.src
        );
    });

    /*
     * Evita que cliques fora do player sejam usados
     * para alterar a página principal.
     */
    window.addEventListener("beforeunload", function () {
        console.log("[PLAYER PROTECAO] página principal protegida.");
    });

})();
</script>
