function toggleMenu(){

    /*
     * CELULAR:
     * mantém exatamente o menu mobile atual.

    if(window.innerWidth <= 768){

        document
            .getElementById("mobileMenu")
            .classList

        return;

        menu.classList.contains("show") &&
        !menu.dataset.carregado

        carregarMenuUsuarioDesktop();

async function carregarMenuUsuarioDesktop(){

    const menu =
        document.getElementById("desktopUserMenu");

    const opcoes =
        document.getElementById("desktopUserOptions");

    if(!menu || !opcoes){
        return;
    }

    try{

        const resposta =
            await fetch(
                "/api/eu",
                {
                    credentials: "same-origin"
                }
            );

        if(!resposta.ok){
            return;
        }

        const dados =
            await resposta.json();

        if(
            !dados ||
            !dados.logado ||
            !dados.usuario
        ){
            return;
        }

        const tipo =
            String(
                dados.usuario.tipo || ""
            ).toLowerCase();

        if(tipo === "admin"){

            opcoes.innerHTML = '';

            const adminLink =
                document.getElementById("lukafilmesDesktopAdmin");

            if(adminLink){
                adminLink.style.display = "block";
                adminLink.textContent = "⚙️ ADMIN";

                adminLink.onclick = function(evento){
                    evento.preventDefault();
                    evento.stopPropagation();
                    evento.stopImmediatePropagation();
                    window.location.assign("/admin.html");
                    return false;
                };
            }

        }else if(tipo === "revendedor"){

            opcoes.innerHTML =
                '<a href="/revendedor.html">🏪 Revendedor</a>';

        }else{

            opcoes.innerHTML = "";

        }

        menu.dataset.carregado = "true";

    }catch(erro){

        console.warn(
            "[MENU DESKTOP] Erro ao verificar usuário:",
            erro
        );

    }
}
